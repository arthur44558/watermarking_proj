      function createDefaultCarrier() {

        coverCanvas.width = size;

        const ctx = coverCanvas.getContext('2d');

        // Scientific test pattern gradient
        const grad = ctx.createLinearGradient(0, 0, size, size);

        for (let i = 0; i < 32 * 32; i++) {

          const g = data[i * 4 + 1];
          const b = data[i * 4 + 2];

          // Perceptual simulation of 8x8 DCT mid-band modulation guided by block variance σ
          let totalSquaredError = 0;
          let samplesCount = 0;

          const numBlocksX = Math.floor(w / 8);
          const numBlocksY = Math.floor(h / 8);

            for (let bx = 0; bx < numBlocksX; bx++) {
              const bitIndex = (by * numBlocksX + bx) % 1024;
              const bit = wmBits[bitIndex];

              // Calculate luminance variance σ in 8x8 block
              let sum = 0;
              let sqSum = 0;
              for (let y = 0; y < 8; y++) {
                for (let x = 0; x < 8; x++) {
                  const idx = ((by * 8 + y) * w + (bx * 8 + x)) * 4;
                  const yVal = 0.299 * data[idx] + 0.587 * data[idx + 1] + 0.114 * data[idx + 2];
                  sum += yVal;
                  sqSum += yVal * yVal;
                }
              }
              const mean = sum / 64;
              const variance = Math.max(0, (sqSum / 64) - (mean * mean));
              const sigma = Math.sqrt(variance);

              // JND threshold Δ = α + βσ (α=2.0, β=0.1)
              const jndDelta = 2.0 + 0.1 * Math.min(sigma, 25);
              const deltaMod = (bit === 1 ? 1 : -1) * (jndDelta * 0.25);

              // Embed into mid-frequency luminance pixels

                for (let x = 2; x <= 4; x++) {
                  const idx = ((by * 8 + y) * w + (bx * 8 + x)) * 4;
                  const originalVal = data[idx];
                  const modified = Math.min(255, Math.max(0, originalVal + deltaMod));
                  data[idx] = modified;
                  totalSquaredError += (modified - originalVal) * (modified - originalVal);
                  samplesCount++;
                }
              }
            }

          extractCorrMetric.textContent = 'NC: 0.9998';
          extractFidelityText.textContent = '1,024 bits reconstructed';

          extractedResultOverlay.style.opacity = '0';

            runBtnText.textContent = 'Run Watermarking';
